
CREATE TABLE User (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    role ENUM('student','tutor','professor','admin') NOT NULL
);

CREATE TABLE Student (
    student_id INT PRIMARY KEY,
    semester INT,
    cgpa DECIMAL(3,2),
    department VARCHAR(50),
    FOREIGN KEY (student_id) REFERENCES User(user_id)
);

CREATE TABLE Tutor (
    tutor_id INT PRIMARY KEY,
    FOREIGN KEY (tutor_id) REFERENCES Student(student_id)
);

CREATE TABLE Professor (
    professor_id INT PRIMARY KEY,
    designation VARCHAR(50),
    FOREIGN KEY (professor_id) REFERENCES User(user_id)
);

CREATE TABLE Admin (
    admin_id INT PRIMARY KEY,
    FOREIGN KEY (admin_id) REFERENCES User(user_id)
);

CREATE TABLE Course (
    course_id VARCHAR(10) PRIMARY KEY,
    course_name VARCHAR(100) NOT NULL,
    credit INT
);

CREATE TABLE QualifiesFor (
    eligibility_id INT AUTO_INCREMENT PRIMARY KEY,
    tutor_id INT NOT NULL,
    course_id VARCHAR(10) NOT NULL,
    grade_obtained VARCHAR(2) NOT NULL CHECK (grade_obtained IN ('A', 'A+')),
    approval_status ENUM('pending','approved','rejected') DEFAULT 'pending',
    approval_date DATE,
    FOREIGN KEY (tutor_id) REFERENCES Tutor(tutor_id),
    FOREIGN KEY (course_id) REFERENCES Course(course_id),
    UNIQUE(tutor_id, course_id)
);

CREATE TABLE Availability (
    slot_id INT AUTO_INCREMENT PRIMARY KEY,
    tutor_id INT NOT NULL,
    day ENUM('Mon','Tue','Wed','Thu','Fri','Sat','Sun') NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    FOREIGN KEY (tutor_id) REFERENCES Tutor(tutor_id)
);

CREATE TABLE Session (
    session_id INT AUTO_INCREMENT PRIMARY KEY,
    student_id INT NOT NULL,
    tutor_id INT NOT NULL,
    course_id VARCHAR(10) NOT NULL,
    session_date DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    status ENUM('pending','accepted','rejected','completed') DEFAULT 'pending',
    FOREIGN KEY (student_id) REFERENCES Student(student_id),
    FOREIGN KEY (tutor_id) REFERENCES Tutor(tutor_id),
    FOREIGN KEY (course_id) REFERENCES Course(course_id)
);

CREATE TABLE Review (
    review_id INT AUTO_INCREMENT PRIMARY KEY,
    student_id INT NOT NULL,
    tutor_id INT NOT NULL,
    rating INT NOT NULL CHECK (rating BETWEEN 1 AND 5),
    comment TEXT,
    review_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (student_id) REFERENCES Student(student_id),
    FOREIGN KEY (tutor_id) REFERENCES Tutor(tutor_id),
    UNIQUE(student_id, tutor_id)
);

CREATE TABLE StudyGroup (
    group_id INT AUTO_INCREMENT PRIMARY KEY,
    subject VARCHAR(100) NOT NULL,
    max_members INT DEFAULT 4,
    created_by INT NOT NULL,
    FOREIGN KEY (created_by) REFERENCES Student(student_id)
);

CREATE TABLE GroupMember (
    group_id INT NOT NULL,
    student_id INT NOT NULL,
    join_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (group_id, student_id),
    FOREIGN KEY (group_id) REFERENCES StudyGroup(group_id),
    FOREIGN KEY (student_id) REFERENCES Student(student_id)
);

CREATE TABLE RevisionSession (
    revision_id INT AUTO_INCREMENT PRIMARY KEY,
    course_id VARCHAR(10) NOT NULL,
    exam_date DATE NOT NULL,
    session_date DATETIME NOT NULL,
    topic VARCHAR(100),
    min_students INT DEFAULT 3,
    status ENUM('pending','confirmed','cancelled') DEFAULT 'pending',
    FOREIGN KEY (course_id) REFERENCES Course(course_id)
);

CREATE TABLE RevisionRequest (
    request_id INT AUTO_INCREMENT PRIMARY KEY,
    revision_id INT NOT NULL,
    student_id INT NOT NULL,
    request_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (revision_id) REFERENCES RevisionSession(revision_id),
    FOREIGN KEY (student_id) REFERENCES Student(student_id),
    UNIQUE(revision_id, student_id)
);

CREATE TABLE Recommendation (
    rec_id INT AUTO_INCREMENT PRIMARY KEY,
    professor_id INT NOT NULL,
    tutor_id INT NOT NULL,
    course_id VARCHAR(10) NOT NULL,
    note TEXT,
    rec_date DATETIME DEFAULT CURRENT_TIMESTAMP
);
